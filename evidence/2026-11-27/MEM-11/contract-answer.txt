def entries(db: Session, member_id: int, *, limit: int = 200) -> list[LedgerEntry]:
    return list(
        db.scalars(
            select(LedgerEntry)
            .where(LedgerEntry.member_id == member_id)
            .order_by(LedgerEntry.id.desc())
            .limit(limit)
        )
    )


def balance(db: Session, member_id: int) -> int:
    return int(
        db.scalar(
            select(func.coalesce(func.sum(LedgerEntry.amount_rial), 0)).where(
                LedgerEntry.member_id == member_id
            )
        )
        or 0
    )

## نتیجه
- `GET /payments/ledger/{member_id}` → `balance_rial` + `reconcile{wallet_balance_rial, ledger_balance_rial, matches, difference_rial}` + `entries[]`.
- `entries` با `id desc` و `limit=200` و **بدون offset** ⇒ صفحه‌بندی سمت سرور وجود ندارد؛ UI باید همین را بگوید نه اینکه صفحه بسازد (تست gate دارد).
- مبالغ **عدد صحیح ریال**‌اند ⇒ هر نمایش باید واحد «ریال» داشته باشد؛ بدون واحد ممنوع (قاعدهٔ MEM-06/07).
- `kind` ∈ charge|pay|refund|reversal ، به‌همراه `source_type`/`source_id`/`reverses_entry_id`/`reversed_by_entry_id`.
- `matches=false` ⇒ اختلاف کیف پول و دفتر؛ UI باید هشدار بدهد نه پنهان‌کردن (تست gate دارد).
- موجودی از `balance_rial` سرور خوانده می‌شود، نه جمع کلاینت.
